local pages = {}
local currentPage

local function makePage(name)
    local page = Instance.new("Frame")
    page.Name = name
    page.Size = UDim2.fromScale(1, 1)
    page.BackgroundTransparency = 1
    page.Visible = false
    page.Parent = content
    pages[name] = page
    return page
end

local function switchPage(name)
    for pageName, page in pairs(pages) do
        page.Visible = pageName == name
    end
    currentPage = name
end

local function makeButton(parent, text, position, size, callback)
    local button = Instance.new("TextButton")
    button.Size = size or UDim2.new(1, 0, 0, 38)
    button.Position = position or UDim2.new()
    button.BackgroundColor3 = BUTTON
    button.BorderSizePixel = 0
    button.AutoButtonColor = false
    button.Font = Enum.Font.GothamMedium
    button.Text = text
    button.TextColor3 = WHITE
    button.TextSize = 12
    button.Parent = parent
    Instance.new("UICorner", button).CornerRadius = UDim.new(0, 9)

    button.MouseEnter:Connect(function()
        TweenService:Create(button, TweenInfo.new(0.15), {BackgroundColor3 = ACCENT}):Play()
    end)
    button.MouseLeave:Connect(function()
        TweenService:Create(button, TweenInfo.new(0.15), {BackgroundColor3 = BUTTON}):Play()
    end)
    button.MouseButton1Click:Connect(callback)
    return button
end

local function getCharacter()
    return player.Character or player.CharacterAdded:Wait()
end

local function removeCosmetic(name)
    local character = player.Character
    if not character then return end
    local old = character:FindFirstChild(name)
    if old then old:Destroy() end
end

local function addHandheldProp(name, bladeColor, handleColor, isHammer)
    removeCosmetic(name)
    local character = getCharacter()
    local hand = character:FindFirstChild("RightHand") or character:FindFirstChild("Right Arm")
    if not hand then return end

    local model = Instance.new("Model")
    model.Name = name
    model.Parent = character

    local function part(partName, size, color, offset, material)
        local p = Instance.new("Part")
        p.Name = partName
        p.Size = size
        p.Color = color
        p.Material = material or Enum.Material.Metal
        p.CanCollide = false
        p.CanTouch = false
        p.CanQuery = false
        p.Massless = true
        p.Anchored = false
        p.Parent = model
        p.CFrame = hand.CFrame * offset
        local weld = Instance.new("WeldConstraint")
        weld.Part0 = hand
        weld.Part1 = p
        weld.Parent = p
        return p
    end

    if isHammer then
        part("Handle", Vector3.new(0.16, 1.5, 0.16), handleColor,
            CFrame.new(0, -0.2, -0.55) * CFrame.Angles(math.rad(15), 0, math.rad(10)),
            Enum.Material.Wood)
        part("HammerHead", Vector3.new(0.85, 0.48, 0.48), bladeColor,
            CFrame.new(0, 0.55, -0.55) * CFrame.Angles(math.rad(15), 0, math.rad(10)),
            Enum.Material.Metal)
    else
        part("Handle", Vector3.new(0.18, 0.55, 0.18), handleColor,
            CFrame.new(0, -0.25, -0.55) * CFrame.Angles(math.rad(15), 0, math.rad(10)),
            Enum.Material.Wood)
        part("Guard", Vector3.new(0.5, 0.09, 0.2), handleColor,
            CFrame.new(0, 0.08, -0.55) * CFrame.Angles(math.rad(15), 0, math.rad(10)))
        part("Blade", Vector3.new(0.12, 2.0, 0.28), bladeColor,
            CFrame.new(0, 1.12, -0.55) * CFrame.Angles(math.rad(15), 0, math.rad(10)),
            Enum.Material.Metal)
    end
end

local function clearSkin()
    local character = player.Character
    if not character then return end
    for _, obj in ipairs(character:GetChildren()) do
        if obj:IsA("BasePart") and obj:GetAttribute("VereusOriginalColor") then
            obj.Color = obj:GetAttribute("VereusOriginalColor")
            obj:SetAttribute("VereusOriginalColor", nil)
        end
    end
end

local function applySkin(color)
    local character = getCharacter()
    for _, obj in ipairs(character:GetChildren()) do
        if obj:IsA("BasePart") and obj.Name ~= "HumanoidRootPart" then
            if not obj:GetAttribute("VereusOriginalColor") then
                obj:SetAttribute("VereusOriginalColor", obj.Color)
            end
            obj.Color = color
        end
    end
end

-- Drag the panel using the header.
local dragging = false
local dragStart, startPos
top.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        dragging = true
        dragStart = input.Position
        startPos = main.Position
        input.Changed:Connect(function()
            if input.UserInputState == Enum.UserInputState.End then dragging = false end
        end)
    end
end)
top.InputChanged:Connect(function(input)
    if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
        local delta = input.Position - dragStart
        main.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X,
            startPos.Y.Scale, startPos.Y.Offset + delta.Y)
    end
end)
